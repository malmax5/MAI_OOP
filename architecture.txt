-----Equipment-----
!Base!
class Material: private_fields: name, armor_multiple
                constructor(name)
                public_funcs: GetName, GetArmorMultiple, SetArmorMultiple, SetName, GetArmorMultiple

class Equip:    private_fields: armor, material, name
                public_func: GetArmor, SetArmor, GetMaterial, SetMaterial, SetName, GetName

!Derived Equip!
class Helmet:   implements: Equip

class Chestplate: implements: Equip

class Boots: implements: Equip

!Derived Material!
class Gold:  implements Material
             base_constructor: SetArmorMultiple, SetName
class Iron: implements Material
             base_constructor: SetArmorMultiple, SetName
class Dimond: implements Material
             base_constructor: SetArmorMultiple, SetName

!Usage!
class MyEquip:  private_fields: Equip[3](0 - Helmet, 1 - Chestplate, 2 - Boots), additionArmor
                public_funcs: SetHelmet, SetChestplate, SetBoots
                Events: TotalArmor -> additionArmor

<Material Builder>
class(interface) MaterialBuilder:  clear_virtual_public_funcs: Reset, SetMaterialName, SetMaterialArmorMultiple
class MaterialBuilderMain:  public_funcs: BuildIronMaterial, BuildGoldMaterial, BuildDimondMaterial

<Equip Builder>
class(interface) EquipBuilder: clear_virtual_public_funcs: Reset, SetArmor, SetMaterial
class EquipBuilderMain: public_funcs: 

-----Mob-----
!Base!
class Positionable :    fields: xCord, yCord
                        funcs: GetX, SetX, GetY, SetY

class NPC:  implements: Positionable
            fields: npc_type_id, current_npc_id, hp, armor, equip, speed
            funcs: set_npc_id, set_current_npc_id, set_hp, set_armor, set_speed, <-getters
            
class AgressiveNPC: implements: NPC
                    fields: attack_damage, attack_distance, target

class PeacefulNPC: implements: NPC

!Derived!
class Knight: implements: AgressiveNPC
class Squirrel: implements: AgressiveNPC
class Pegasus: implements: PeacefulNPC
